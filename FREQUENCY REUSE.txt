
import matplotlib.pyplot as plt
from matplotlib.patches import Polygon, Patch
import numpy as np

R = 1.0  # Radius of hexagon

def hex_corner(center, size, k):
    """Calculate 6 vertices for a flat-topped hexagon."""
    angle_rad = np.pi / 180 * (60 * k)
    return (
        center[0] + size * np.cos(angle_rad),
        center[1] + size * np.sin(angle_rad)
    )

def axial_to_pixel(q, r, size):
    """Convert axial coordinates to Cartesian coordinates."""
    x = size * (3 / 2 * q)
    y = size * (np.sqrt(3) / 2 * q + np.sqrt(3) * r)
    return x, y


N_TO_SHIFT = {
    1: (1, 0),
    3: (1, 1),
    4: (2, 0),
    7: (2, 1),
    9: (3, 0),
    12: (2, 2),
    13: (3, 1),
    19: (3, 2)
}

# Different colors for different clusters
cluster_colors = [
    '#AED6F1',  # Light Blue
    '#F9E79F',  # Light Yellow
    '#ABEBC6',  # Light Green
    '#F5B7B1',  # Light Red
    '#D7BDE2',  # Light Purple
    '#FAD7A0',  # Light Orange
    '#A3E4D7',  # Light Cyan
    '#D5DBDB'   # Light Gray
]

def generate_cluster_template(N):
    """Generate local cell coordinates for a cluster."""
    i_shift, j_shift = N_TO_SHIFT.get(N, (2, 1))
    radius_search = i_shift + j_shift + 1

    coords = []

    for q in range(-radius_search, radius_search + 1):
        for r in range(-radius_search, radius_search + 1):
            coords.append((q, r))

    coords.sort(
        key=lambda c: (
            c[0]**2 + c[1]**2 + (c[0] + c[1])**2
        )
    )

    template = []

    for idx in range(min(N, len(coords))):
        q, r = coords[idx]
        template.append((q, r, f"F{idx + 1}"))

    return template, i_shift, j_shift


print("--- Cellular Frequency Reuse Setup ---")
print("Valid cluster sizes: 1, 3, 4, 7, 9, 12, 13, 19")

try:
    N_input = int(
        input("Enter number of cells in a cluster (e.g., 7): ") or 7
    )

    num_clusters = int(
        input("Enter number of clusters (e.g., 2): ") or 2
    )

except ValueError:
    N_input, num_clusters = 7, 2

if N_input not in N_TO_SHIFT:
    print("Invalid cluster size. Defaulting to N=7.")
    N_input = 7

num_clusters = max(1, num_clusters)

cluster_template, i_shift, j_shift = generate_cluster_template(N_input)

# Generate cluster origins
cluster_origins = [(0, 0)]

shift_vectors = [
    (i_shift, j_shift),
    (-j_shift, i_shift + j_shift),
    (-i_shift - j_shift, i_shift),
    (-i_shift, -j_shift),
    (j_shift, -i_shift - j_shift)
]

for idx in range(1, num_clusters):
    vec = shift_vectors[(idx - 1) % len(shift_vectors)]
    prev_origin = cluster_origins[-1]

    cluster_origins.append((
        prev_origin[0] + vec[0],
        prev_origin[1] + vec[1]
    ))

# Plot
fig, ax = plt.subplots(figsize=(12, 9))

f1_centers = []
legend_patches = []

for c_idx, origin in enumerate(cluster_origins):

    oq, or_ = origin

    # Assign a different color to each cluster
    color = cluster_colors[c_idx % len(cluster_colors)]

    legend_patches.append(
        Patch(facecolor=color, edgecolor='black',
              label=f'Cluster {c_idx + 1}')
    )

    for dq, dr, freq in cluster_template:

        q, r = oq + dq, or_ + dr
        cx, cy = axial_to_pixel(q, r, R)

        if freq == 'F1':
            f1_centers.append((cx, cy))

        vertices = [
            hex_corner((cx, cy), R, k)
            for k in range(6)
        ]

        is_f1 = (freq == 'F1')

        hex_patch = Polygon(
            vertices,
            facecolor=color,
            edgecolor='black',
            linewidth=2.5 if is_f1 else 1.0,
            zorder=2
        )

        ax.add_patch(hex_patch)

        # Frequency labels
        ax.text(
            cx, cy, freq,
            ha='center',
            va='center',
            fontsize=10,
            fontweight='bold' if is_f1 else 'normal',
            color='black',
            zorder=4
        )

    # Cluster label
    c_x, c_y = axial_to_pixel(oq, or_, R)

    ax.text(
        c_x, c_y - 0.5,
        f"Cluster {c_idx + 1}",
        ha='center',
        va='center',
        fontsize=9,
        fontweight='bold',
        fontstyle='italic',
        bbox=dict(
            boxstyle='round,pad=0.2',
            facecolor='white',
            edgecolor='gray',
            alpha=0.9
        ),
        zorder=5
    )

# Draw co-channel reuse lines
if len(f1_centers) >= 2:

    for idx in range(len(f1_centers) - 1):

        p1 = f1_centers[idx]
        p2 = f1_centers[idx + 1]

        ax.plot(
            [p1[0], p2[0]],
            [p1[1], p2[1]],
            color='red',
            linestyle='--',
            linewidth=2.5,
            marker='o',
            markersize=6,
            label='Co-channel Reuse Line (F1)' if idx == 0 else "",
            zorder=6
        )

# Graph formatting
ax.set_aspect('equal')
ax.autoscale_view()
ax.axis('off')

plt.title(
    f"Cellular Frequency Reuse "
    f"({num_clusters} Clusters, N = {N_input} Cells Each)",
    fontsize=14,
    fontweight='bold',
    pad=20
)

# Cluster color legend
ax.legend(
    handles=legend_patches + [
        plt.Line2D(
            [0], [0],
            color='red',
            linestyle='--',
            linewidth=2,
            label='Co-channel Reuse (F1)'
        )
    ],
    loc='upper left',
    bbox_to_anchor=(1.02, 1),
    frameon=True
)

plt.tight_layout()
plt.show()
